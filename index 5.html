let activeDecks = {};
let currentCategory = '';
let score = 0;
let gameMode = 'time';
let timeLeft = 60;
let timerInterval;
const screens = {
menu: document.getElementById('menu-screen'),
prep: document.getElementById('prep-screen'),
game: document.getElementById('game-screen'),
result: document.getElementById('result-screen')
};
const ui = {
word: document.getElementById('current-word'),
score: document.getElementById('final-score'),
tapLeft: document.getElementById('tap-left'),
tapRight: document.getElementById('tap-right'),
gameWrapper: document.getElementById('game-screen'),
btnFinish: document.getElementById('btn-finish'),
timer: document.getElementById('timer'),
prepTimer: document.getElementById('prep-timer'),
resultTitle: document.getElementById('result-title')
};
function switchScreen(name) {
Object.values(screens).forEach(s => s.classList.remove('active'));
screens[name].classList.add('active');
}
function setMode(mode) {
gameMode = mode;
document.getElementById('mode-time').className = mode === 'time' ? 'flex-1 py-3 rounded-xl font-bold bg-gray-600 text-white transition' : 'flex-1 py-3 rounded-xl font-bold text-gray-400 hover:text-white transition';
document.getElementById('mode-free').className = mode === 'free' ? 'flex-1 py-3 rounded-xl font-bold bg-gray-600 text-white transition' : 'flex-1 py-3 rounded-xl font-bold text-gray-400 hover:text-white transition';
}
function shuffle(array) {
let cur = array.length, rnd;
while (cur !== 0) {
rnd = Math.floor(Math.random() * cur);
cur--;
[array[cur], array[rnd]] = [array[rnd], array[cur]];
}
return array;
}
function startGame(category) {
currentCategory = category;
score = 0;
if (!activeDecks[category] || activeDecks[category].length === 0) {
activeDecks[category] = shuffle([...databases[category]]);
}
ui.timer.style.opacity = gameMode === 'time' ? '1' : '0';
switchScreen('prep');
let prepTime = 5;
ui.prepTimer.innerText = prepTime;
const prepInterval = setInterval(() => {
prepTime--;
ui.prepTimer.innerText = prepTime;
if (prepTime <= 0) {
clearInterval(prepInterval);
startRound();
}
}, 1000);
}
function startRound() {
switchScreen('game');
nextWord();
if (gameMode === 'time') {
timeLeft = 60;
ui.timer.innerText = timeLeft;
timerInterval = setInterval(() => {
timeLeft--;
ui.timer.innerText = timeLeft;
if (timeLeft <= 0) {
endGame('Время вышло!');
}
}, 1000);
}
}
function nextWord() {
if (activeDecks[currentCategory].length === 0) {
activeDecks[currentCategory] = shuffle([...databases[currentCategory]]);
}
ui.word.innerText = activeDecks[currentCategory].pop();
}
function endGame(reason = 'Конец игры!') {
clearInterval(timerInterval);
ui.resultTitle.innerText = reason;
ui.score.innerText = score;
switchScreen('result');
}
function triggerFlash(colorClass) {
ui.gameWrapper.classList.remove('flash-green', 'flash-red');
void ui.gameWrapper.offsetWidth;
ui.gameWrapper.classList.add(colorClass);
setTimeout(() => ui.gameWrapper.classList.remove(colorClass), 300);
}
function handleTapLeft(e) {
e.preventDefault();
triggerFlash('flash-red');
nextWord();
}
function handleTapRight(e) {
e.preventDefault();
score++;
triggerFlash('flash-green');
nextWord();
}
function handleFinish(e) {
e.preventDefault();
e.stopPropagation();
endGame('Игра завершена');
}
ui.tapLeft.addEventListener('touchstart', handleTapLeft);
ui.tapLeft.addEventListener('mousedown', handleTapLeft);
ui.tapRight.addEventListener('touchstart', handleTapRight);
ui.tapRight.addEventListener('mousedown', handleTapRight);
ui.btnFinish.addEventListener('touchstart', handleFinish);
ui.btnFinish.addEventListener('mousedown', handleFinish);
